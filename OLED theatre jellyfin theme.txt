/* ── performance.css ─────────────────────────────────────────── */
/*
 * Abyss Cinematic Edition — performance.css
 * GPU-compositing hints, will-change budgets, contain declarations,
 * and strategies to keep the theme fast on all devices.
 *
 * Rules:
 *  - Only transform + opacity get animated (GPU-native)
 *  - box-shadow and filter are never transitioned on large elements
 *  - will-change is applied sparingly — it costs VRAM
 *  - contain: layout/style on scroll containers prevents reflow cascade
 */

/* ─── GPU layer hints: applied only to frequently animated elements ─── */

/* Cards animate on hover — reserve a GPU layer per row,
   not per card (cost-effective) */
.emby-scroller,
.itemsContainer.scrollX {
  contain: layout style !important;
  will-change: scroll-position;
}

/* Backdrop is always animating */
.backdropImage,
.itemBackdropImage {
  will-change: transform;
  transform: translateZ(0);       /* force compositing */
  backface-visibility: hidden;
  -webkit-backface-visibility: hidden;
}

/* Cards hoist to GPU only on hover — avoids wasting VRAM at rest */
.card:hover .cardContent {
  will-change: transform;
  transform: translateZ(0);
}

/* Dialog animations */
.dialog {
  will-change: transform, opacity, filter;
  transform: translateZ(0);
}

/* OSD is overlaid & frequently faded */
.videoOsdBottom,
.videoOsd,
.videoOsdTop {
  will-change: opacity;
  transform: translateZ(0);
}

}

.verticalSection,
.homeSection {

}

.dialog {
